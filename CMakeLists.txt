cmake_minimum_required(VERSION 3.21)
project(kanjoos VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(KNJ_ENABLE_HIP "Build the real ROCm/HIP backend" OFF)
option(KNJ_BUILD_TESTS "Build correctness and stress tests" ON)
option(KNJ_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
option(KNJ_SANITIZE "Enable address/undefined-behaviour sanitizers" OFF)
set(KNJ_GPU_ARCHS "gfx1031;gfx1201" CACHE STRING "Native GPU targets (no HSA override)")
find_package(Threads REQUIRED)
add_library(knj_options INTERFACE)
target_include_directories(knj_options INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_include_directories(knj_options SYSTEM INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
if(MSVC)
  target_compile_options(knj_options INTERFACE /W4 /permissive- /fp:strict)
  if(KNJ_WARNINGS_AS_ERRORS)
    target_compile_options(knj_options INTERFACE /WX)
  endif()
else()
  target_compile_options(knj_options INTERFACE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Wno-missing-field-initializers;-ffp-contract=off>)
  if(KNJ_WARNINGS_AS_ERRORS)
    target_compile_options(knj_options INTERFACE $<$<COMPILE_LANGUAGE:CXX>:-Werror>)
  endif()
endif()
include(cmake/sanitize.cmake)
add_library(knj_substrate STATIC src/util/sha256.cpp src/platform/platform.cpp src/gguf/gguf_reader.cpp src/gguf/model_index.cpp)
target_link_libraries(knj_substrate PUBLIC knj_options Threads::Threads)
if(WIN32)
  target_link_libraries(knj_substrate PUBLIC psapi)
endif()
add_library(knj_tensor STATIC src/tensor/tensor_io.cpp)
target_link_libraries(knj_tensor PUBLIC knj_substrate)
add_library(knj_router STATIC src/router/router.cpp)
target_link_libraries(knj_router PUBLIC knj_tensor)
add_library(knj_compute STATIC src/compute/reference.cpp)
target_link_libraries(knj_compute PUBLIC knj_options)
add_library(knj_quant STATIC src/compiler/quant.cpp)
target_link_libraries(knj_quant PUBLIC knj_compute knj_substrate)
add_library(knj_store STATIC src/store/expert_store.cpp)
target_link_libraries(knj_store PUBLIC knj_quant knj_router)
add_library(knj_runtime STATIC
  src/device/backend.cpp src/profile/profiler.cpp src/core/runtime.cpp
  src/core/config.cpp src/core/budget.cpp src/core/autotuner.cpp
  src/host/pools.cpp src/gpu/slot_pool.cpp src/transfer/engine.cpp
  src/residency/admission.cpp src/residency/predictor.cpp src/residency/manager.cpp
  src/compute/executor.cpp src/kv/cache.cpp src/kv/radix.cpp src/attn/attention.cpp)
target_link_libraries(knj_runtime PUBLIC knj_store)
if(KNJ_ENABLE_HIP)
  include(cmake/rocm.cmake)
  include(cmake/arch.cmake)
  add_library(knj_hip STATIC src/device/hip_backend.hip kernels/ops.hip)
  set_source_files_properties(src/device/hip_backend.hip kernels/ops.hip PROPERTIES LANGUAGE HIP)
  set_target_properties(knj_hip PROPERTIES HIP_ARCHITECTURES "${KNJ_GPU_ARCHS}")
  target_include_directories(knj_hip PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/kernels)
  target_link_libraries(knj_hip PUBLIC knj_compute knj_substrate hip::host)
  target_compile_options(knj_hip PRIVATE $<$<COMPILE_LANGUAGE:HIP>:${KNJ_HIP_WAVE_FLAGS};-ffp-contract=off;-fno-fast-math>)
  target_link_libraries(knj_runtime PRIVATE knj_hip)
  target_compile_definitions(knj_runtime PRIVATE KNJ_WITH_HIP=1)
endif()
# Pinned, vendored canonical GGUF tokenizer and Jinja interpreter. Never
# create a llama context or load inference weights through this dependency.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static third-party dependencies" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/llama EXCLUDE_FROM_ALL)
add_library(knj_jinja STATIC
  third_party/llama/common/json.cpp third_party/llama/common/jinja/lexer.cpp
  third_party/llama/common/jinja/parser.cpp third_party/llama/common/jinja/runtime.cpp
  third_party/llama/common/jinja/string.cpp third_party/llama/common/jinja/value.cpp
  third_party/llama/common/unicode.cpp)
target_include_directories(knj_jinja SYSTEM PUBLIC third_party/llama/common PRIVATE third_party/llama/src third_party)
target_link_libraries(knj_jinja PUBLIC llama)
add_library(knj_tokenizer STATIC src/tokenizer/tokenizer.cpp)
target_link_libraries(knj_tokenizer PUBLIC knj_options knj_substrate PRIVATE knj_jinja)
add_library(knj_model STATIC src/model/model.cpp)
target_link_libraries(knj_model PUBLIC knj_runtime knj_tokenizer)
add_library(knj_inference STATIC src/inference/sampler.cpp src/spec/controller.cpp src/inference/engine.cpp)
target_link_libraries(knj_inference PUBLIC knj_model)
add_executable(kanjoos-inspect src/tools/kanjoos_inspect.cpp)
target_link_libraries(kanjoos-inspect PRIVATE knj_substrate)
add_executable(kanjoos-store src/tools/kanjoos_store.cpp)
target_link_libraries(kanjoos-store PRIVATE knj_store)
if(KNJ_BUILD_TESTS)
  enable_testing()
  foreach(name substrate router store runtime kv compiler)
    if(name STREQUAL "store" AND WIN32)
      continue() # POSIX process-death harness; portable store tests run in engine suite.
    endif()
    add_executable(test_${name} tests/unit/test_${name}.cpp)
    target_include_directories(test_${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests/unit)
    target_link_libraries(test_${name} PRIVATE knj_runtime)
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} TIMEOUT 90)
  endforeach()
  add_executable(test_model tests/unit/test_model.cpp)
  target_include_directories(test_model PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests/unit)
  target_link_libraries(test_model PRIVATE knj_inference)
  add_test(NAME model COMMAND test_model)
  set_tests_properties(model PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} TIMEOUT 600)
endif()
install(TARGETS kanjoos-inspect kanjoos-store RUNTIME DESTINATION bin)
install(FILES kanjoos.toml DESTINATION share/kanjoos)
